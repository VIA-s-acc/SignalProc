cmake_minimum_required(VERSION 3.28)
project(signals VERSION 0.7.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(SIGNALS_SANITIZE "Build own targets with AddressSanitizer and UBSan" OFF)
option(SIGNALS_WERROR "Treat warnings in own targets as errors" OFF)
option(SIGNALS_BUILD_TESTS "Build the test suite" ON)
set(SIGNALS_VERSION_SUFFIX "" CACHE STRING "Pre-release suffix appended to the project version, e.g. -rc1")

if(MSVC)
  add_compile_options(/utf-8 /bigobj /Zc:__cplusplus)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
endif()

include(cmake/Warnings.cmake)
include(cmake/Dependencies.cmake)

add_subdirectory(src/core)
add_subdirectory(src/io)
add_subdirectory(src/svg)
add_subdirectory(src/audio)
add_subdirectory(src/i18n)
add_subdirectory(src/theory)
add_subdirectory(src/pipeline)
add_subdirectory(src/learn)
add_subdirectory(src/tui)
add_subdirectory(src/commands)
add_subdirectory(app)

if(CMAKE_BUILD_TYPE STREQUAL "Release")
  add_subdirectory(bench)
endif()

if(SIGNALS_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
